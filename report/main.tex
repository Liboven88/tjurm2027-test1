\section{my\_strlen：自己实现字符串长度}
\subsection{题目}
% TODO：用自己的话复述要求（别看源码，凭理解写）
统计一个以 \texttt{'\textbackslash 0'} 结尾的字符串有多少个字符，不调用现成的
\texttt{strlen}。

\subsection{思路}
% TODO：一两句话。指针怎么走、什么时候停
从首地址开始，只要当前字符不是结束符就往后走一步并计数；遇到结束符时计数值就是长度。

\subsection{代码}
\begin{lstlisting}
int my_strlen(char* str) {
  int len = 0;
  while (str[len] != '\0') ++len;
  return len;
}
\end{lstlisting}

\subsection{结果}
% TODO：换成你的终端输出
\texttt{test\_strlen()} 输出：\textbf{通过}（对 \texttt{"123456"}、\texttt{""}、
\texttt{"hello world!"} 三个样例都与标准库结果一致）。

\section{my\_strcat：字符串拼接}
\subsection{题目}
% TODO
把传入的两个字符串首尾拼接在一起，组成一个新的字符串，并确保以\texttt{'\textbackslash 0'}结尾
\subsection{思路}
% TODO
让传进来的*str1走到str1的结束符上，再利用循环把str2的字符值逐个赋给*str1，二者每轮循环自增，最后在拼接字符串末尾加上\texttt{'\textbackslash 0'}，过程中要保证str1申请的内存空间足够大
\subsection{代码}
\begin{lstlisting}
void my_strcat (char* str_1, char* str_2){
  while (*str_1 != '\0') str_1++;
  while (*str_2 != '\0') {
    *str_1 = *str_2;
    str_1++;
    str_2++;
  }
  *str_1 = '\0';
}
\end{lstlisting}
\subsection{结果}
% TODO
\texttt{test\_strcat()} 输出：\textbf{通过}（先用标准库 \texttt{strcat} 算出期望结果，
再用 \texttt{strcmp} 比较整串，说明结束符 \texttt{'\textbackslash 0'} 也写对了）。

\section{my\_strstr：子串查找}
\subsection{题目}
% TODO
在一个字符串中查找是否有与目标子串完全相同的部分，若有，返回这部分的首地址；若无，返回nullptr
\subsection{思路}
% TODO
通过循环找出第一个相同的字符，并将之后的每个字符进行对比，若完全相同则返回第一个相同字符在字符串中的地址；若出现不同则break，寻找下一个“第一个相同的字符”,一直没找到返回nullptr。过程中需保证字符串排查余下的长度大于等于字串长度
特殊情况：目标子串为空时，直接返回字符串首地址
\subsection{代码}
\begin{lstlisting}
char* my_strstr(char* s, char* p) {
  int len_s = my_strlen(s), len_p = my_strlen(p);
  if (len_p == 0) return s;
  for (int i = 0; i < len_s - len_p + 1; ++i) {
    for (int j = 0; j < len_p; ++j) {
      if (s[i + j] != p[j]) break;
      if (j == len_p - 1) return &s[i];
    }
  }
  return 0;
}
\end{lstlisting}
\subsection{结果}
% TODO
\texttt{test\_strstr()} 输出：\textbf{通过}（对 \texttt{"wjsfjdvioz"}、\texttt{"qqqqq"}、
\texttt{"j"} 三个子串，返回的指针都与标准库 \texttt{strstr} 相同）。

\section{rgb2gray：彩色图转灰度图}
\subsection{题目}
% TODO
将传入的彩色图转为灰度图
\subsection{思路}
% TODO：记得写清下标公式
彩色图在内存里按 \texttt{R,G,B} 三通道连续存放，第 $(x,y)$ 个像素的三个通道位于
\texttt{in[(y*w+x)*3]}、\texttt{+1}、\texttt{+2}。灰度公式为
\begin{equation}
V = 0.2989\,R + 0.5870\,G + 0.1140\,B
\end{equation}
以纵坐标自增为外层循环，横坐标自增为内层循环，遍历图片的每个像素，并用灰度公式进行转化
\subsection{代码}
\begin{lstlisting}
void rgb2gray(float* in, float* out, int h, int w) {
  for (int y = 0; y < h; ++y){
    for (int x = 0; x < w; ++x){
      int index = y*w + x;
      out[index] = in[index*3]*0.2989 + in[index*3 + 1]*0.5870 + in[index*3 + 2]*0.1140;
    }
  }
}
\end{lstlisting}

\section{resize：双线性插值缩放}
\subsection{题目}
% TODO
通过双线性插值法，实现图片的缩放函数
\subsection{思路}
% TODO
\begin{equation}
             x2 - x          x - x1
    v = v1 * ———————— + v2 * ————————
             x2 - x1         x2 - x1
\end{equation}
1. 对于每个 dst 中的像素点 (x, y)，先计算出其在 src 中的坐标 float(x0,y0)，
2. 然后计算出其在 src 中的四个邻居点:
 x1 = static\_cast<int>(x0), y1 = static\_cast<int>(y0)
  上面这样可以直接将 float 通过下取整的方式转换为 int， 剩下三个邻居就好找了
3. 用双线性插值法，用四个邻居的v值算出插值，再赋给 dst 中的像素点
4. 注意上面的方法中，四个邻居点的坐标可能会超出 src 的范围， 所以需要对其进行边界检查
\subsection{代码}
\begin{lstlisting}
void resize(float* in, float* out, int h, int w, int c, float scale) {
  int new_h = h * scale, new_w = w * scale;
  for (int y = 0; y < new_h; ++y){
    for (int x = 0; x < new_w; ++x){
      for (int ch = 0; ch < c; ++ch){
  float x0 = static_cast<float>(x) / scale, y0 = static_cast<float>(y) / scale;
  int x1 = static_cast<int>(x0), int y1 = static_cast<int>(y0);
  if(x1 >= w - 1) x1 = w - 2;
  if(y1 >= h - 1) y1 = h - 2;
  int x2 = x1 + 1, int y2 = y1 + 1;
  float q1 = in[(y1*w + x1)*c + ch] * (x2 - x0) / (x2 - x1)
           + in[(y1*w + x2)*c + ch] * (x0 - x1) / (x2 - x1);
  float q2 = in[(y2*w + x1)*c + ch] * (x2 - x0) / (x2 - x1)
           + in[(y2*w + x2)*c + ch] * (x0 - x1) / (x2 - x1);
           float Q = q1 * (y2 - y0) / (y2 - y1) + q2 * (y0 - y1) / (y2 - y1);
           out[(y*new_w + x)*c + ch] = Q; 
      }
    }
  }
}
\end{lstlisting}

\section{hist\_eq：直方图均衡化}
\subsection{题目}
% TODO
让传入的灰度图片灰度级分布更均匀，即增强图片对比度
\subsection{思路}
% TODO
1.建好数桶，记录每个灰度数出现的频次
2.累计每个灰度级及其以下的灰度级的总频次
3.依据每个灰度级及其以下的灰度级的总频次在总像素数的占比，映射出新的均衡的灰度级分布
4.把新的灰度级赋给对应像素
\subsection{代码}
\begin{lstlisting}

void hist_eq(float* in, int h, int w) {
  int hist[256] = {0};
  for (int i = 0; i < h*w; ++i) ++hist[static_cast<int>(in[i])];
  int cdf[256] = {0};
  cdf[0] = hist[0];
  for (int i = 1; i < 256; ++i) cdf[i] = cdf[i - 1] + hist[i];
  float map[256] = {0};
  for (int i = 0; i < 256; ++i) map[i] = static_cast<float>(cdf[i]) / static_cast<float>(h * w) * 255.0f;;
  for (int i = 0; i < h*w; ++i) in[i] = map[static_cast<int>(in[i])];
}
\end{lstlisting}
